   479→/* ----------------------------- Monthly trend ------------------------------ */
   480→
   481→function MonthlyTrend({
   482→  data,
   483→}: {
   484→  data: Array<{ key: string; label: string; count: number; points: number }>
   485→}) {
   486→  const fa = useFaDigits()
   487→  const W = 320
   488→  const H = 140
   489→  const PAD_X = 18
   490→  const PAD_TOP = 12
   491→  const PAD_BOTTOM = 28
   492→  const plotW = W - PAD_X * 2
   493→  const plotH = H - PAD_TOP - PAD_BOTTOM
   494→
   495→  const max = Math.max(1, ...data.map((d) => d.points))
   496→  const min = 0
   497→  const span = max - min || 1
   498→  const n = data.length
   499→
   500→  const xAt = (i: number) => PAD_X + (n <= 1 ? plotW / 2 : (i * plotW) / (n - 1))
   501→  const yAt = (v: number) => PAD_TOP + plotH - ((v - min) / span) * plotH
   502→
   503→  const linePath = data
   504→    .map((d, i) => `${i === 0 ? 'M' : 'L'} ${xAt(i).toFixed(2)} ${yAt(d.points).toFixed(2)}`)
   505→    .join(' ')
   506→  const areaPath = `${linePath} L ${xAt(n - 1).toFixed(2)} ${PAD_TOP + plotH} L ${xAt(0).toFixed(2)} ${PAD_TOP + plotH} Z`
   507→
   508→  const gridLines = [0, 0.25, 0.5, 0.75, 1]
   509→
   510→  return (
   511→    <Card className="py-0">
   512→      <CardHeader className="pb-2">
   513→        <CardTitle className="text-base">روند ماهانه</CardTitle>
   514→      </CardHeader>
   515→      <CardContent className="pb-3">
   516→        <svg
   517→          viewBox={`0 0 ${W} ${H}`}
   518→          className="h-auto w-full"
   519→          role="img"
   520→          aria-label="نمودار روند ماهانه امتیازها"
   521→          preserveAspectRatio="none"
   522→        >
   523→          <defs>
   524→            <linearGradient id="insights-area" x1="0" y1="0" x2="0" y2="1">
   525→              <stop offset="0%" stopColor="#10b981" stopOpacity="0.35" />
   526→              <stop offset="100%" stopColor="#10b981" stopOpacity="0.02" />
   527→            </linearGradient>
   528→            <linearGradient id="insights-line" x1="0" y1="0" x2="1" y2="0">
   529→              <stop offset="0%" stopColor="#10b981" />
   530→              <stop offset="100%" stopColor="#f59e0b" />
   531→            </linearGradient>
   532→          </defs>
   533→
   534→          {/* Gridlines */}
   535→          {gridLines.map((g, i) => {
   536→            const y = PAD_TOP + plotH * g
   537→            return (
   538→              <line
   539→                key={i}
   540→                x1={PAD_X}
   541→                y1={y}
   542→                x2={W - PAD_X}
   543→                y2={y}
   544→                stroke="currentColor"
   545→                strokeOpacity={0.08}
   546→                strokeWidth={1}
   547→              />
   548→            )
   549→          })}
   550→
   551→          {/* Area fill */}
   552→          {data.length > 1 && <path d={areaPath} fill="url(#insights-area)" />}
   553→          {/* Line */}
   554→          {data.length > 0 && (
   555→            <path
   556→              d={linePath}
   557→              fill="none"
   558→              stroke="url(#insights-line)"
   559→              strokeWidth={2}
   560→              strokeLinejoin="round"
   561→              strokeLinecap="round"
   562→            />
   563→          )}
   564→
   565→          {/* Data points */}
   566→          {data.map((d, i) => (
   567→            <g key={d.key}>
   568→              <circle
   569→                cx={xAt(i)}
   570→                cy={yAt(d.points)}
   571→                r={3.5}
   572→                fill="#ffffff"
   573→                stroke="#10b981"
   574→                strokeWidth={2}
   575→              />
   576→              <text
   577→                x={xAt(i)}
   578→                y={H - 10}
   579→                textAnchor="middle"
   580→                className="fill-muted-foreground"
   581→                style={{ fontSize: 10 }}
   582→              >
   583→                {d.label}
   584→              </text>
   585→            </g>
   586→          ))}
   587→        </svg>
   588→        {data.every((d) => d.count === 0) && (
   589→          <div className="text-muted-foreground mt-1 text-center text-xs">
   590→            در این بازه خریدی ثبت نشده است.
   591→          </div>
   592→        )}
   593→        <div className="text-muted-foreground mt-1 flex items-center justify-between px-1 text-xs tabular-nums">
   594→          <span>بیشینه: {fa(max)} امتیاز</span>
   595→          <span>۶ ماه اخیر</span>
   596→        </div>
   597→      </CardContent>
   598→    </Card>
   599→  )
   600→}
